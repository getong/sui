4//! Direct mapping from `tx_sequence_number → (TransactionDigest, event_count, tx_offset, checkpoint_number)`.
5//!
6//! Row key layout: `[bit_reversed_tx_seq_be_u64]` — 8 bytes. Bit reversal
7//! spreads monotonic transaction sequence numbers across the keyspace while
8//! preserving a reversible point-lookup key.
9//!
10//! `event_count` lets readers enumerate a transaction's event_seqs without
11//! reading the tx row itself — used by unfiltered event listing to discover
12//! exactly the events contributing to a page.
13//!
14//! `tx_offset` is the transaction's zero-based position within its checkpoint.
15//! It lets readers report a `(checkpoint, tx_offset)` coordinate without an
16//! extra `checkpoints` lookup to recover the checkpoint's first tx_seq. The
17//! column is assumed always present (the table is backfilled before serving),
18//! so `decode` treats a missing `tx_offset` as an error.
19
20use anyhow::bail;
21use anyhow::{Context, Result};
22use bytes::Bytes;
23use integer_encoding::VarInt;
24use sui_types::digests::TransactionDigest;
25use sui_types::messages_checkpoint::CheckpointSequenceNumber;
26
27pub const NAME: &str = "tx_seq_digest";
28
29pub mod col {
30    /// Raw 32-byte TransactionDigest.
31    pub const DIGEST: &str = "d";
32    /// Big-endian u32 count of events emitted by this transaction.
33    pub const EVENT_COUNT: &str = "e";
34    /// Varint-encoded zero-based position of this transaction within its checkpoint.
35    pub const TX_OFFSET: &str = "o";
36    /// Varint-encoded checkpoint sequence number containing this transaction.
37    pub const CHECKPOINT_NUMBER: &str = "c";
38}
39
40/// Row key: `tx_seq.reverse_bits().to_be_bytes()`.
41pub fn encode_key(tx_seq: u64) -> Vec<u8> {
42    tx_seq.reverse_bits().to_be_bytes().to_vec()
43}
44
45/// Decode a bit-reversed row key.
46pub fn decode_key(key: &[u8]) -> Result<u64> {
47    if key.len() != 8 {
48        anyhow::bail!("tx_seq_digest key not 8 bytes (got {})", key.len());
49    }
50    Ok(u64::from_be_bytes(key.try_into().unwrap()).reverse_bits())
51}
52
53pub fn encode(
54    digest: &TransactionDigest,
55    event_count: u32,
56    tx_offset: u32,
57    checkpoint_number: CheckpointSequenceNumber,
58) -> [(&'static str, Bytes); 4] {
59    [
60        (col::DIGEST, Bytes::from(digest.inner().to_vec())),
61        (
62            col::EVENT_COUNT,
63            Bytes::copy_from_slice(&event_count.to_be_bytes()),
64        ),
65        (col::TX_OFFSET, Bytes::from(tx_offset.encode_var_vec())),
66        (
67            col::CHECKPOINT_NUMBER,
68            Bytes::from(checkpoint_number.encode_var_vec()),
69        ),
70    ]
71}
72
73pub fn decode(
74    cells: &[(Bytes, Bytes)],
75) -> Result<(TransactionDigest, u32, u32, CheckpointSequenceNumber)> {
76    let mut digest: Option<TransactionDigest> = None;
77    let mut event_count: u32 = 0;
78    let mut tx_offset = None;
79    let mut checkpoint_number = None;
80    for (column, value) in cells {
81        if column.as_ref() == col::DIGEST.as_bytes() {
82            let bytes: [u8; 32] = value
83                .as_ref()
84                .try_into()
85                .context("tx_seq_digest digest not 32 bytes")?;
86            digest = Some(TransactionDigest::from(bytes));
87        } else if column.as_ref() == col::EVENT_COUNT.as_bytes() {
88            let bytes: [u8; 4] = value
89                .as_ref()
90                .try_into()
91                .context("tx_seq_digest event_count not 4 bytes")?;
92            event_count = u32::from_be_bytes(bytes);
93        } else if column.as_ref() == col::TX_OFFSET.as_bytes() {
94            tx_offset = Some(decode_canonical_varint::<u32>(value, "tx_offset")?);
95        } else if column.as_ref() == col::CHECKPOINT_NUMBER.as_bytes() {
96            checkpoint_number = Some(decode_checkpoint_number_value(value)?);
97        }
98    }
99    Ok((
100        digest.context("tx_seq_digest missing digest column")?,
101        event_count,
102        tx_offset.context("tx_seq_digest missing tx_offset column")?,
103        checkpoint_number.context("tx_seq_digest missing checkpoint_number column")?,
104    ))
105}
106
107pub fn decode_checkpoint_number(cells: &[(Bytes, Bytes)]) -> Result<CheckpointSequenceNumber> {
108    for (column, value) in cells {
109        if column.as_ref() == col::CHECKPOINT_NUMBER.as_bytes() {
110            return decode_checkpoint_number_value(value);
111        }
112    }
113
114    bail!("tx_seq_digest missing checkpoint_number column")
115}
116
117fn decode_checkpoint_number_value(value: &Bytes) -> Result<CheckpointSequenceNumber> {
118    decode_canonical_varint::<u64>(value, "checkpoint_number")
119}
120
121/// Decode a canonical varint cell value, rejecting trailing or non-minimal
122/// bytes so each value has exactly one valid encoding.
123fn decode_canonical_varint<T: VarInt>(value: &Bytes, field: &str) -> Result<T> {
124    let (decoded, bytes_read) = T::decode_var(value.as_ref())
125        .with_context(|| format!("tx_seq_digest {field} is not a valid varint"))?;
126    if bytes_read != value.len() {
127        bail!("tx_seq_digest {field} has trailing bytes");
128    }
129    if decoded.required_space() != bytes_read {
130        bail!("tx_seq_digest {field} is not canonical");
131    }
132    Ok(decoded)
133}

135#[cfg(test)]
136mod tests {
137    use super::*;
138
139    #[test]
140    fn encode_key_layout() {
141        for tx_seq in [0u64, 1, 63, 64, 65, 1_000_000, u64::MAX] {
142            let k = encode_key(tx_seq);
143            assert_eq!(k.len(), 8, "key must be 8 bytes for tx_seq={tx_seq}");
144            assert_eq!(&k, &tx_seq.reverse_bits().to_be_bytes());
145        }
146    }
147
148    #[test]
149    fn encode_decode_round_trip() {
150        for tx_seq in [0u64, 1, 63, 64, 65, 1_000_000, u64::MAX] {
151            let k = encode_key(tx_seq);
152            let got = decode_key(&k).expect("decode must accept freshly encoded key");
153            assert_eq!(got, tx_seq);
154        }
155    }
156
157    #[test]
158    fn encode_decode_row_round_trip() {
159        let digest = TransactionDigest::new([7; 32]);
160        let event_count = 123_456;
161        let tx_offset = 42;
162        let checkpoint_number = 300_000_000;
163        let cells = encode(&digest, event_count, tx_offset, checkpoint_number);
164
165        assert_eq!(cells[1].1.as_ref(), &event_count.to_be_bytes());
166        assert_eq!(cells[2].1.as_ref(), tx_offset.encode_var_vec().as_slice());
167        assert_eq!(cells[3].1.len(), 5);
168
169        let cells = cells
170            .into_iter()
171            .map(|(column, value)| (Bytes::from_static(column.as_bytes()), value))
172            .collect::<Vec<_>>();
173        let (decoded_digest, decoded_event_count, decoded_tx_offset, decoded_checkpoint_number) =
174            decode(&cells).unwrap();
175        assert_eq!(decoded_digest, digest);
176        assert_eq!(decoded_event_count, event_count);
177        assert_eq!(decoded_tx_offset, tx_offset);
178        assert_eq!(decoded_checkpoint_number, checkpoint_number);
179        assert_eq!(decode_checkpoint_number(&cells).unwrap(), checkpoint_number);
180    }
181
182    /// The table is backfilled before serving, so a row without the
183    /// `tx_offset` column is treated as corruption rather than defaulting.
184    #[test]
185    fn decode_rejects_missing_tx_offset() {
186        let digest = TransactionDigest::new([3; 32]);
187        let checkpoint_number = 7;
188        let cells = vec![
189            (
190                Bytes::from_static(col::DIGEST.as_bytes()),
191                Bytes::from(digest.inner().to_vec()),
192            ),
193            (
194                Bytes::from_static(col::EVENT_COUNT.as_bytes()),
195                Bytes::copy_from_slice(&5u32.to_be_bytes()),
196            ),
197            (
198                Bytes::from_static(col::CHECKPOINT_NUMBER.as_bytes()),
199                Bytes::from(checkpoint_number.encode_var_vec()),
200            ),
201        ];
202        assert!(decode(&cells).is_err());
203    }
204
205    #[test]
206    fn decode_key_rejects_wrong_length() {
207        assert!(decode_key(&[0u8; 7]).is_err());
208        assert!(decode_key(&[0u8; 9]).is_err());
209        assert!(decode_key(&[]).is_err());
210    }
