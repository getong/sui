4//! DNF bitmap index queries over ordered bucket streams.
5//!
6//! Callers build a `BitmapQuery` as an OR of terms. Each term is an AND of
7//! signed dimension-key literals. Evaluation yields matching bitmap members as
8//! they are produced. Back-pressure from downstream consumers, e.g. a
9//! `.take(page_size)`, propagates back to the backend-provided bucket streams
10//! and avoids materializing matches we won't use.
11//!
12//! Queries are intentionally restricted to anchored DNF: every term must contain
13//! at least one positive literal. Positive literals give the evaluator concrete
14//! bitmap streams to scan and intersect; negative literals only shrink those
15//! candidate streams. Supporting negative-only terms such as `NOT sender = A`
16//! would require scanning an external universe for the requested range and
17//! subtracting from it, which defeats the index's selective streaming behavior
18//! and makes pagination depend on a full-range scan. Requiring DNF at the API
19//! boundary also keeps this evaluator as a set of ordered stream merge-joins
20//! instead of a recursive expression engine or a query normalizer.
21//!
22//! Backends provide one ordered `(bucket_id, RoaringBitmap)` stream per
23//! dimension key. The merge-join machinery here is storage-agnostic: BigTable,
24//! RocksDB, or any other backend can reuse it as long as its bucket stream is
25//! sparse, ordered by the requested scan direction, and stores bitmap positions
26//! relative to that bucket.
27

45/// Terminal signal: the per-request bucket-fetch budget is exhausted.
46/// Surfaced as `anyhow::Error` so it short-circuits `try_stream!`
47/// pipelines through the existing error path. A silent EOF would let
48/// `subtract_two` emit unfiltered include rows past the exclude side's
49/// last fetched bucket.
50#[derive(Debug, thiserror::Error)]
51#[error("bitmap scan budget exhausted")]
52pub struct BitmapScanLimitExceeded;

54/// Aggregate of multiple terminal errors raised during the same poll
55/// cycle of a multi-way combinator (e.g. two children of `intersect_n`
56/// both error before the next loop iteration). The single-error
57/// shortcut returns the inner error directly so existing downcasts on
58/// the wire keep working — `MultiError` only appears when 2+ errors
59/// coincide, in which case downstream consumers should use
60/// [`error_contains`] to interrogate the aggregate.
61#[derive(Debug)]
62pub struct MultiError(Vec<anyhow::Error>);
63
64impl MultiError {
65    /// Wrap a non-empty error list into an `anyhow::Error`. With exactly
66    /// one error, returns it unwrapped so the common case preserves
67    /// `downcast_ref` behavior on the original concrete type.
68    pub fn collapse(mut errs: Vec<anyhow::Error>) -> anyhow::Error {
69        assert!(!errs.is_empty(), "MultiError::collapse on empty Vec");
70        if errs.len() == 1 {
71            return errs.pop().expect("len == 1");
72        }
73        anyhow::Error::new(MultiError(errs))
74    }
75
76    pub fn iter(&self) -> impl Iterator<Item = &anyhow::Error> {
77        self.0.iter()
78    }
79}
80
81impl std::fmt::Display for MultiError {
82    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
83        write!(f, "{} concurrent errors", self.0.len())?;
84        for (i, e) in self.0.iter().enumerate() {
85            write!(f, "\n  [{i}] {e}")?;
86        }
87        Ok(())
88    }
89}

93/// Downcast probe that looks through a `MultiError` aggregate. Returns
94/// the first inner `T` whether the top-level error is `T` directly or a
95/// `MultiError` containing one. Use this instead of `downcast_ref::<T>`
96/// at sites that may receive aggregated errors from the bitmap
97/// combinators.
98pub fn error_contains<T: std::error::Error + Send + Sync + 'static>(
99    err: &anyhow::Error,
100) -> Option<&T> {
101    if let Some(t) = err.downcast_ref::<T>() {
102        return Some(t);
103    }
104    if let Some(multi) = err.downcast_ref::<MultiError>() {
105        return multi.iter().find_map(|e| e.downcast_ref::<T>());
106    }
107    None
108}

110/// Item or progress watermark flowing through a bitmap eval pipeline.
111/// `Watermark(p)` means every Item with position strictly before `p`
112/// in scan direction has been emitted upstream. Downstream stages must
113/// preserve watermark/item ordering — that's what makes the watermark a
114/// safe resume cursor on timeout.
115#[derive(Clone, Copy, Debug, Eq, PartialEq)]
116pub enum Watermarked<T> {
117    Item(T),
118    Watermark(u64),
119}
120
121impl<T> Watermarked<T> {
122    pub fn map_item<U>(self, f: impl FnOnce(T) -> U) -> Watermarked<U> {
123        match self {
124            Watermarked::Item(t) => Watermarked::Item(f(t)),
125            Watermarked::Watermark(p) => Watermarked::Watermark(p),
126        }
127    }
128}

130/// Per-request bucket-scan accounting, delivered via the `on_metrics`
131/// callback passed to `eval_bitmap_query_stream`. Fires once when the
132/// eval pipeline is dropped (natural end, error, or consumer cancel).
133/// The sole exception is the budget-misconfig early-out, which errors
134/// before any scan is set up and emits nothing.
135#[derive(Clone, Copy, Debug, Eq, PartialEq)]
136pub struct BitmapScanMetrics {
137    /// Buckets actually evaluated (charged against the per-request
138    /// budget). At exhaustion each leaf may have fetched one extra
139    /// bucket that was discarded rather than evaluated, so observed
140    /// backend reads can exceed this by up to `BitmapQuery::leaf_count()`.
141    pub buckets_evaluated: u64,
142}
143
144/// Per-request evaluated-bucket budget shared across all dimension
145/// streams of one eval. Charges are post-poll — see
146/// `budget_limited_bucket_stream`.
147#[derive(Clone)]
148struct BitmapScanBudget {
149    initial: u64,
150    remaining: Arc<AtomicU64>,
151}
152
153impl BitmapScanBudget {
154    fn new(initial: u64) -> Self {
155        Self {
156            initial,
157            remaining: Arc::new(AtomicU64::new(initial)),
158        }
159    }
160
161    /// Charge one bucket. Returns false on underflow.
162    fn try_take(&self) -> bool {
163        self.remaining
164            .fetch_update(Ordering::SeqCst, Ordering::SeqCst, |b| {
165                if b == 0 { None } else { Some(b - 1) }
166            })
167            .is_ok()
168    }
169
170    /// Charge a leaf's mandatory first bucket: decrements the shared pool
171    /// when it can, but ALWAYS succeeds. The runtime guards `budget >=
172    /// leaf_count`, but a *shared* atomic with concurrent leaves gives no
173    /// ordering guarantee — a sparse term can drain the pool before a
174    /// slower sibling leaf charges its first bucket, leaving that leaf's
175    /// merged-watermark slot `None` forever (cursorless `SCAN_LIMIT`).
176    /// Reserving the first bucket per leaf makes the `leaf_count` floor's
177    /// promise — "every leaf emits its first watermark" — actually hold.
178    /// Charging-when-possible keeps `buckets_evaluated` accurate in the
179    /// common `budget >> leaves` case; it only undercounts a first bucket
180    /// taken after the pool was already exhausted by other leaves.
181    fn take_first(&self) {
182        let _ = self
183            .remaining
184            .fetch_update(Ordering::SeqCst, Ordering::SeqCst, |b| {
185                Some(b.saturating_sub(1))
186            });
187    }
188
189    fn buckets_evaluated(&self) -> u64 {
190        self.initial
191            .saturating_sub(self.remaining.load(Ordering::SeqCst))
192    }
193}

195/// Merge per-child watermark slots into the combinator's output.
196/// `None` if any child has yet to emit a first watermark — combinator
197/// must not claim progress past an unknown floor. Ascending min,
198/// descending max (the slowest source is the floor).
199fn merge_watermarks(slots: &[Option<u64>], direction: ScanDirection) -> Option<u64> {
200    // Short-circuits to None if any slot is None.
201    let all: Vec<u64> = slots.iter().copied().collect::<Option<Vec<_>>>()?;
202    if all.is_empty() {
203        return None;
204    }
205    Some(match direction {
206        ScanDirection::Ascending => *all.iter().min().expect("non-empty"),
207        ScanDirection::Descending => *all.iter().max().expect("non-empty"),
208    })
209}

211/// The less-advanced of two frontier positions in scan direction: the min
212/// ascending, the max descending. Used to keep a merged frontier bounded by
213/// the slowest contributor.
214fn bound_in_direction(a: u64, b: u64, direction: ScanDirection) -> u64 {
215    match direction {
216        ScanDirection::Ascending => a.min(b),
217        ScanDirection::Descending => a.max(b),
218    }
219}

221/// Like [`merge_watermarks`] but also folds in a `ceiling` frozen from
222/// errored (retired) union branches. A retired branch no longer polls, but
223/// the position it reached still bounds how far the union can claim progress,
224/// so it stays in the merge as a fixed floor. Live slots dominate as usual
225/// (any `None` live slot blocks progress); with no live slots left the
226/// ceiling alone is the frontier.
227fn merge_with_ceiling(
228    slots: &[Option<u64>],
229    ceiling: Option<u64>,
230    direction: ScanDirection,
231) -> Option<u64> {
232    // A live branch that has not yet reported blocks progress past its
233    // unknown floor.
234    if slots.iter().any(Option::is_none) {
235        return None;
236    }
237    let live = match direction {
238        ScanDirection::Ascending => slots.iter().flatten().copied().min(),
239        ScanDirection::Descending => slots.iter().flatten().copied().max(),
240    };
241    match (live, ceiling) {
242        (Some(l), Some(c)) => Some(bound_in_direction(l, c, direction)),
243        (Some(l), None) => Some(l),
244        (None, c) => c,
245    }
246}

248/// RAII guard: fires `on_metrics` exactly once on drop with the final
249/// `BitmapScanMetrics`. Held inside the boxed eval stream so the callback
250/// fires on natural end, error, or consumer cancel.
251struct ObserveOnDrop<F: FnOnce(BitmapScanMetrics) + Send + 'static> {
252    budget: BitmapScanBudget,
253    callback: Option<F>,
254}
255
256impl<F: FnOnce(BitmapScanMetrics) + Send + 'static> Drop for ObserveOnDrop<F> {
257    fn drop(&mut self) {
258        if let Some(cb) = self.callback.take() {
259            cb(BitmapScanMetrics {
260                buckets_evaluated: self.budget.buckets_evaluated(),
261            });
262        }
263    }
264}

375impl BitmapQuery {
376    pub fn new(terms: Vec<BitmapTerm>) -> Result<Self> {
377        if terms.is_empty() {
378            bail!("bitmap query must contain at least one term");
379        }
380        Ok(Self { terms })
381    }

383    pub fn scan(dimension_key: Vec<u8>) -> Result<Self> {
384        Ok(Self {
385            terms: vec![BitmapTerm::new(vec![BitmapLiteral::include(

391    /// Total leaf literal count across every term. The per-request budget

394    /// every leaf emits its first watermark regardless of how the shared
395    /// budget races; see `BitmapScanBudget::take_first`.
396    pub fn leaf_count(&self) -> usize {
397        self.terms.iter().map(|t| t.literals.len()).sum()
398    }
399}

413/// Evaluate a DNF `BitmapQuery` against a backend-provided bitmap source.
414///
415/// `budget` caps evaluated buckets across all dimension scans (see
416/// [`BitmapScanLimitExceeded`] and [`BitmapScanMetrics`]). `on_metrics`
417/// fires exactly once when the eval stream is dropped.
418///
419/// Output emits `Watermarked::Item(absolute_member_id)` interleaved with
420/// `Watermarked::Watermark(p)` merged from the leaf streams — sparse
421/// intersects that drop every bucket still emit watermarks at the rate
422/// sources advance.
423pub fn eval_bitmap_query_stream<S, F>(
424    source: S,
425    query: BitmapQuery,
426    range: Range<u64>,
427    bucket_size: u64,
428    direction: ScanDirection,
429    budget: u64,
430    on_metrics: F,
431) -> BoxStream<'static, Result<Watermarked<u64>>>
432where
433    S: BitmapBucketSource,
434    F: FnOnce(BitmapScanMetrics) + Send + 'static,
435{
436    let leaves = query.leaf_count();
437    if (budget as usize) < leaves {
438        // Misconfig guard: short-circuit before any scan setup. No
439        // `on_metrics` here — there's no scan to account for, and the
440        // error surfaces on its own. `on_metrics` is dropped uncalled.
441        return async_stream::stream! {
442            yield Err(anyhow::anyhow!(
443                "bitmap scan budget {budget} is insufficient for {leaves} leaf streams; \
444                 server is misconfigured"
445            ));
446        }
447        .boxed();
448    }
449    let budget = BitmapScanBudget::new(budget);
450    let bucket_stream = eval_bitmap_query_bucket_stream(
451        source,
452        query,
453        range.clone(),
454        bucket_size,
455        direction,
456        budget.clone(),
457    );
458    let inner = flatten_watermarked_buckets(bucket_stream, range.clone(), bucket_size, direction);
459    // Wrapping the guard inside `async_stream::stream!` keeps it alive
460    // for the stream's full lifetime; the callback fires when the
461    // consumer drops the returned `BoxStream`.
462    let guard = ObserveOnDrop {
463        budget,
464        callback: Some(on_metrics),
465    };
466    let range_terminus = if direction.is_ascending() {
467        range.end
468    } else {
469        range.start
470    };
471    async_stream::stream! {
472        let _guard = guard;
473        futures::pin_mut!(inner);
474        let mut last_emitted: Option<u64> = None;
475        while let Some(item) = inner.next().await {
476            let is_err = item.is_err();
477            if let Ok(Watermarked::Watermark(p)) = &item {
478                last_emitted = Some(*p);
479            }
480            yield item;
481            if is_err {
482                // Don't synthesize a final range-terminus watermark on
483                // error — the scan did NOT reach it.
484                return;
485            }
486        }
487        // Natural EOF: guarantee a final range-terminus watermark so
488        // clients learn "scan covered the entire range." Combinators that
489        // short-circuit early (e.g. intersect when one side EOFs) leave
490        // `last_emitted` short of the terminus; this fills the gap.
491        if frontier_advanced(last_emitted, range_terminus, direction) {
492            yield Ok(Watermarked::Watermark(range_terminus));
493        }
494    }
495    .boxed()
496}

498fn frontier_advanced(prev: Option<u64>, next: u64, direction: ScanDirection) -> bool {
499    match prev {
500        None => true,
501        Some(prev) => match direction {
502            ScanDirection::Ascending => next > prev,
503            ScanDirection::Descending => next < prev,
504        },
505    }
506}

508/// Evaluate a DNF `BitmapQuery` as an ordered `WatermarkedBucketStream`.
509fn eval_bitmap_query_bucket_stream<S>(
510    source: S,
511    query: BitmapQuery,
512    range: Range<u64>,
513    bucket_size: u64,
514    direction: ScanDirection,
515    budget: BitmapScanBudget,
516) -> WatermarkedBucketStream
517where
518    S: BitmapBucketSource,
519{
520    let streams: Vec<WatermarkedBucketStream> = query
521        .terms
522        .into_iter()
523        .map(|term| {
524            term_bucket_stream(
525                source.clone(),
526                term,
527                range.clone(),
528                bucket_size,
529                direction,
530                budget.clone(),
531            )
532            .boxed()
533        })
534        .collect();
535    union_n(streams, direction).boxed()
536}

538/// Evaluate one DNF term: intersect all includes, then subtract excludes.
539fn term_bucket_stream<S>(
540    source: S,
541    term: BitmapTerm,
542    range: Range<u64>,
543    bucket_size: u64,
544    direction: ScanDirection,
545    budget: BitmapScanBudget,
546) -> WatermarkedBucketStream
547where
548    S: BitmapBucketSource,
549{
550    let mut include_streams: Vec<WatermarkedBucketStream> = Vec::new();
551    let mut exclude_streams: Vec<WatermarkedBucketStream> = Vec::new();
552
553    for literal in term.literals {
554        let (key, is_include) = match literal {
555            BitmapLiteral::Include(key) => (key.into_inner(), true),
556            BitmapLiteral::Exclude(key) => (key.into_inner(), false),
557        };
558        let stream = budget_limited_bucket_stream(
559            source.scan_bucket_stream(key, range.clone(), direction),
560            budget.clone(),
561            range.clone(),
562            bucket_size,
563            direction,
564        )
565        .boxed();
566        if is_include {
567            include_streams.push(stream);
568        } else {
569            exclude_streams.push(stream);
570        }
571    }
572
573    let include_stream = intersect_n(include_streams, direction).boxed();
574    // Skip subtract when nothing to exclude. `union_n` of zero streams
575    // emits nothing, which would leave `subtract_two`'s `b_watermark`
576    // slot `None` forever and suppress every merged watermark.
577    if exclude_streams.is_empty() {
578        return include_stream;
579    }
580    let exclude_stream = union_n(exclude_streams, direction);
581
582    // `subtract_two` polls both sides with `try_join!`, so include and
583    // exclude scans are opened/read concurrently.
584    subtract_two(include_stream, exclude_stream, direction).boxed()
585}

587/// Wrap a per-dimension raw bucket stream into a `WatermarkedBucketStream`,
588/// emitting `Watermark(pre), Item, Watermark(post)` per bucket and a
589/// final `Watermark(range_terminus)` on natural EOF.
590///
591/// On budget exhaustion yields `Err(BitmapScanLimitExceeded)` — never
592/// silent EOF, which `subtract_two` would mistake for "no more excludes."
593/// Charge is post-poll so `budget == leaves` doesn't spuriously trip at
594/// natural EOF.
595///
596/// The post-poll timing also underwrites forward progress under
597/// `SCAN_LIMIT`: a bucket's `pre` and `post` are both emitted before the
598/// *next* bucket's charge can trip the limit. So a leaf that scans at
599/// least one bucket always surfaces an *advancing* `post` watermark, not
600/// just the first bucket's `pre` — which is clamped back to `range.start`
601/// (below) and would otherwise resume the client at their own request
602/// floor. Downstream `resolve_watermarks` flushes that in-flight `post`
603/// ahead of the `BitmapScanLimitExceeded`, so a truncated scan still
604/// hands back real progress.
605fn budget_limited_bucket_stream<S>(
606    inner: S,
607    budget: BitmapScanBudget,
608    range: Range<u64>,
609    bucket_size: u64,
610    direction: ScanDirection,
611) -> impl Stream<Item = WatermarkedBucket> + Send + 'static
612where
613    S: Stream<Item = BucketItem> + Send + 'static,
614{
615    let range_terminus = if direction.is_ascending() {
616        range.end
617    } else {
618        range.start
619    };
620    async_stream::try_stream! {
621        futures::pin_mut!(inner);
622        let mut first = true;
623        while let Some(item) = inner.next().await {
624            let item = item?;
625            if first {
626                // Reserved: a leaf's first bucket is always allowed so it
627                // always emits its first watermark, regardless of how the
628                // shared budget raced. See `BitmapScanBudget::take_first`.
629                budget.take_first();
630                first = false;
631            } else if !budget.try_take() {
632                Err(anyhow::Error::new(BitmapScanLimitExceeded))?;
633            }
634            let (bucket_id, _) = &item;
635            let bucket_start = bucket_id.saturating_mul(bucket_size);
636            let bucket_end_exclusive = bucket_start.saturating_add(bucket_size);
637            // Clamp to the request range: cursors round-trip into
638            // subsequent requests with different ranges, so an
639            // out-of-bound cursor would be a foot-gun.
640            let (pre, post) = if direction.is_ascending() {
641                (
642                    bucket_start.max(range.start),
643                    bucket_end_exclusive.min(range.end),
644                )
645            } else {
646                (
647                    bucket_end_exclusive.min(range.end),
648                    bucket_start.max(range.start),
649                )
650            };
651            yield Watermarked::Watermark(pre);
652            yield Watermarked::Item(item);
653            yield Watermarked::Watermark(post);
654        }
655        // Natural EOF: emit a final watermark at the range terminus so
656        // combinators learn this source covered the whole range.
657        yield Watermarked::Watermark(range_terminus);
658    }
659}

661/// Multi-way merge intersection. Emits bucket_ids present in every
662/// child, with the bitwise AND of their bitmaps; drops empty results.
663/// Per-child watermarks drain eagerly each iteration; the min/max
664/// merged across children emits when it advances.
665pub fn intersect_n<S>(
666    streams: Vec<S>,
667    direction: ScanDirection,
668) -> impl Stream<Item = WatermarkedBucket> + Send + 'static
669where
670    S: Stream<Item = WatermarkedBucket> + Send + Unpin + 'static,
671{
672    async_stream::try_stream! {
673        if streams.is_empty() {
674            return;
675        }
676        let mut children: Vec<Peekable<S>> =
677            streams.into_iter().map(|s| s.peekable()).collect();
678        let mut child_watermarks: Vec<Option<u64>> = vec![None; children.len()];
679        let mut last_emitted: Option<u64> = None;
680
681        loop {
682            // Drain pending watermarks concurrently so we don't serialize
683            // on the slowest child. Defer errors until AFTER emitting the
684            // merged watermark — progress up to the error point should
685            // still reach the client.
686            let outcomes = futures::future::join_all(
687                children
688                    .iter_mut()
689                    .map(|child| drain_pending_watermarks(Pin::new(child))),
690            )
691            .await;
692            let mut deferred_errors: Vec<anyhow::Error> = Vec::new();
693            for (i, outcome) in outcomes.into_iter().enumerate() {
694                if let Some(p) = outcome.last_watermark {
695                    child_watermarks[i] = Some(p);
696                }
697                if let Some(e) = outcome.error {
698                    deferred_errors.push(e);
699                }
700            }
701            if let Some(merged) = merge_watermarks(&child_watermarks, direction)
702                && frontier_advanced(last_emitted, merged, direction)
703            {
704                yield Watermarked::Watermark(merged);
705                last_emitted = Some(merged);
706            }
707            if !deferred_errors.is_empty() {
708                Err(MultiError::collapse(deferred_errors))?;
709            }
710
711            // Poll children together so independent backend scans run
712            // concurrently.
713            let Some((peeks, max_bucket)) = complete_peeks(peek_buckets(&mut children).await?)
714            else {
715                break;
716            };
717            let target_bucket = match direction {
718                ScanDirection::Ascending => max_bucket,
719                ScanDirection::Descending => peeks.iter().copied().min().expect("non-empty peeks"),
720            };
721
722            if peeks.iter().all(|&b| b == target_bucket) {
723                // All children at the same bucket: intersect and emit.
724                let mut acc: Option<RoaringBitmap> = None;
725                for child in children.iter_mut() {
726                    let (bid, bitmap) = take_bucket_item(Pin::new(child)).await?;
727                    debug_assert_eq!(bid, target_bucket);
728                    acc = Some(match acc {
729                        None => bitmap,
730                        Some(a) => a & bitmap,
731                    });
732                }
733                let bitmap = acc.expect("children non-empty");
734                if !bitmap.is_empty() {
735                    yield Watermarked::Item((target_bucket, bitmap));
736                }
737            } else {
738                // Sparse bitmap rows encode only non-empty buckets. A
739                // child lagging the alignment target intersects with an
740                // implicit all-zero bitmap at that bucket → empty result.
741                // Consume the lagging bucket and re-peek.
742                for (i, child) in children.iter_mut().enumerate() {
743                    let drop_bucket = match direction {
744                        ScanDirection::Ascending => peeks[i] < target_bucket,
745                        ScanDirection::Descending => peeks[i] > target_bucket,
746                    };
747                    if drop_bucket {
748                        let _ = take_bucket_item(Pin::new(child)).await?;
749                    }
750                }
751            }
752        }
753    }
754}

756/// Multi-way merge union. Emits every bucket_id produced by any child,
757/// with the bitwise OR of bitmaps at that bucket. Per-child watermarks
758/// drain eagerly; min/max merged across live children emits when it
759/// advances.
760///
761/// Flush-on-error: a budget-exhausted child is an `OR` branch that died,
762/// not a reason to discard its siblings. Such a child is *retired* —
763/// removed from the live set, its final position frozen as a
764/// `resume_ceiling`, its error stashed. The union keeps emitting the
765/// surviving branches' already-fetched buckets UP TO that ceiling, then
766/// propagates the error. Buckets at or below the ceiling are inside the
767/// region every branch jointly scanned, so they resume cleanly; buckets
768/// PAST it are withheld — they sit ahead of the resumable frontier (pinned
769/// at the dead branch's floor) and would be re-scanned, hence
770/// double-delivered, on resume. Without this, a dead branch would discard a
771/// live sibling's first-bucket result and pin the cursor at the request
772/// floor (a livelock).
773pub fn union_n<S>(
774    streams: Vec<S>,
775    direction: ScanDirection,
776) -> impl Stream<Item = WatermarkedBucket> + Send + 'static
777where
778    S: Stream<Item = WatermarkedBucket> + Send + Unpin + 'static,
779{
780    async_stream::try_stream! {
781        if streams.is_empty() {
782            return;
783        }
784        let mut children: Vec<Peekable<S>> =
785            streams.into_iter().map(|s| s.peekable()).collect();
786        let mut child_watermarks: Vec<Option<u64>> = vec![None; children.len()];
787        let mut last_emitted: Option<u64> = None;
788        // Frozen floor + stashed errors from retired (errored) branches.
789        // `resume_ceiling` caps how far the surviving branches may be
790        // emitted; `pending_errors` propagate once the flush completes.
791        let mut resume_ceiling: Option<u64> = None;
792        let mut pending_errors: Vec<anyhow::Error> = Vec::new();
793
794        loop {
795            // Drain pending watermarks concurrently to avoid serializing
796            // on the slowest child.
797            let outcomes = futures::future::join_all(
798                children
799                    .iter_mut()
800                    .map(|child| drain_pending_watermarks(Pin::new(child))),
801            )
802            .await;
803
804            // Apply watermarks; retire any child that errored — freeze its
805            // final position into the resume ceiling and stash its error to
806            // propagate once the survivors have been flushed.
807            let mut live_children = Vec::with_capacity(children.len());
808            let mut live_watermarks = Vec::with_capacity(children.len());
809            for ((child, outcome), prev) in children
810                .into_iter()
811                .zip_eq(outcomes)
812                .zip_eq(child_watermarks)
813            {
814                let wm = outcome.last_watermark.or(prev);
815                match outcome.error {
816                    Some(e) => {
817                        pending_errors.push(e);
818                        if let Some(f) = wm {
819                            resume_ceiling = Some(match resume_ceiling {
820                                None => f,
821                                Some(c) => bound_in_direction(c, f, direction),
822                            });
823                        }
824                    }
825                    None => {
826                        live_children.push(child);
827                        live_watermarks.push(wm);
828                    }
829                }
830            }
831            children = live_children;
832            child_watermarks = live_watermarks;
833
834            if let Some(merged) = merge_with_ceiling(&child_watermarks, resume_ceiling, direction)
835                && frontier_advanced(last_emitted, merged, direction)
836            {
837                yield Watermarked::Watermark(merged);
838                last_emitted = Some(merged);
839            }
840
841            // No live branch left to emit from.
842            if children.is_empty() {
843                break;
844            }
845
846            let peeks = peek_buckets(&mut children).await?;
847
848            // Evict exhausted children so their resources (e.g. semaphore
849            // permits) release promptly. The evicted child's last
850            // watermark already drained into `last_emitted`, so dropping
851            // its slot is safe.
852            let mut surviving_children = Vec::with_capacity(children.len());
853            let mut surviving_peeks = Vec::with_capacity(peeks.len());
854            let mut surviving_watermarks = Vec::with_capacity(child_watermarks.len());
855            for ((child, peek), wm) in children
856                .into_iter()
857                .zip_eq(peeks)
858                .zip_eq(child_watermarks.into_iter())
859            {
860                if let Some(b) = peek {
861                    surviving_children.push(child);
862                    surviving_peeks.push(b);
863                    surviving_watermarks.push(wm);
864                }
865            }
866            children = surviving_children;
867            child_watermarks = surviving_watermarks;
868
869            if surviving_peeks.is_empty() {
870                break;
871            }
872            let next_bucket = match direction {
873                ScanDirection::Ascending => *surviving_peeks
874                    .iter()
875                    .min()
876                    .expect("non-empty after evicting None peeks"),
877                ScanDirection::Descending => *surviving_peeks
878                    .iter()
879                    .max()
880                    .expect("non-empty after evicting None peeks"),
881            };
882
883            // Resume-ceiling gate: once a branch has died, withhold any
884            // bucket sitting past its frozen floor — on resume from the
885            // ceiling it would be re-scanned and so re-emitted. A bucket's
886            // leading `pre` watermark (recorded per child during the drain)
887            // is its position; the child at `next_bucket` carries the
888            // direction-min/max one.
889            if let Some(ceil) = resume_ceiling {
890                let next_pre = surviving_peeks
891                    .iter()
892                    .position(|&b| b == next_bucket)
893                    .and_then(|i| child_watermarks[i]);
894                let past_ceiling = match (next_pre, direction) {
895                    (Some(p), ScanDirection::Ascending) => p >= ceil,
896                    (Some(p), ScanDirection::Descending) => p <= ceil,
897                    (None, _) => false,
898                };
899                if past_ceiling {
900                    break;
901                }
902            }
903
904            let mut acc: Option<RoaringBitmap> = None;
905            for (i, child) in children.iter_mut().enumerate() {
906                if surviving_peeks[i] == next_bucket {
907                    let (_, bitmap) = take_bucket_item(Pin::new(child)).await?;
908                    acc = Some(match acc {
909                        None => bitmap,
910                        Some(a) => a | bitmap,
911                    });
912                }
913            }
914            if let Some(bitmap) = acc
915                && !bitmap.is_empty()
916            {
917                yield Watermarked::Item((next_bucket, bitmap));
918            }
919        }
920        if !pending_errors.is_empty() {
921            Err(MultiError::collapse(pending_errors))?;
922        }
923    }
924}

926/// Merge-join subtraction for an anchored negative literal.
927///
928/// A negative literal in a valid term is evaluated as `a AND NOT b`, where `a`
929/// is the already-anchored positive candidate stream. Bitmap subtraction
930/// (`a_bm - b_bm`) has the same truth table without materializing `NOT b` over
931/// the whole range:
932///
933/// ```text
934/// a b | a AND NOT b
935/// 1 0 | 1
936/// 0 1 | 0
937/// 0 0 | 0
938/// 1 1 | 0
939/// ```
940///
941/// For each bucket in `a`, emits `a_bm - b_bm` if `b` has the same bucket,
942/// else emits `a_bm` unchanged. Drops empty results.
943pub fn subtract_two<A, B>(
944    a: A,
945    b: B,
946    direction: ScanDirection,
947) -> impl Stream<Item = WatermarkedBucket> + Send + 'static
948where
949    A: Stream<Item = WatermarkedBucket> + Send + 'static,
950    B: Stream<Item = WatermarkedBucket> + Send + 'static,
951{
952    async_stream::try_stream! {
953        let a = a.peekable();
954        let b = b.peekable();
955        futures::pin_mut!(a);
956        futures::pin_mut!(b);
957        let mut a_watermark: Option<u64> = None;
958        let mut b_watermark: Option<u64> = None;
959        let mut last_emitted: Option<u64> = None;
960
961        loop {
962            // Drain both sides concurrently, emit merged "both past P"
963            // if it advanced. Defer errors until after the emit so
964            // mid-drain progress reaches the client.
965            let (a_outcome, b_outcome) = futures::future::join(
966                drain_pending_watermarks(a.as_mut()),
967                drain_pending_watermarks(b.as_mut()),
968            )
969            .await;
970            if let Some(p) = a_outcome.last_watermark {
971                a_watermark = Some(p);
972            }
973            if let Some(p) = b_outcome.last_watermark {
974                b_watermark = Some(p);
975            }
976            let mut deferred_errors: Vec<anyhow::Error> = Vec::new();
977            if let Some(e) = a_outcome.error {
978                deferred_errors.push(e);
979            }
980            if let Some(e) = b_outcome.error {
981                deferred_errors.push(e);
982            }
983            if let Some(merged) = merge_watermarks(&[a_watermark, b_watermark], direction)
984                && frontier_advanced(last_emitted, merged, direction)
985            {
986                yield Watermarked::Watermark(merged);
987                last_emitted = Some(merged);
988            }
989            if !deferred_errors.is_empty() {
990                Err(MultiError::collapse(deferred_errors))?;
991            }
992
993            // Concurrent peek buffers both head rows; the later `next()`
994            // calls consume them.
995            let (a_peek, b_peek) =
996                futures::try_join!(peek_bucket(a.as_mut()), peek_bucket(b.as_mut()))?;
997            let Some(a_bucket) = a_peek else {
998                return;
999            };
1000
1001            match b_peek {
1002                None => {
1003                    // No more negatives: flush a.
1004                    let (bid, bitmap) = take_bucket_item(a.as_mut()).await?;
1005                    if !bitmap.is_empty() {
1006                        yield Watermarked::Item((bid, bitmap));
1007                    }
1008                }
1009                Some(bb)
1010                    if (direction.is_ascending() && bb > a_bucket)
1011                        || (!direction.is_ascending() && bb < a_bucket) =>
1012                {
1013                    // b is ahead, emit a unchanged.
1014                    let (bid, bitmap) = take_bucket_item(a.as_mut()).await?;
1015                    if !bitmap.is_empty() {
1016                        yield Watermarked::Item((bid, bitmap));
1017                    }
1018                }
1019                Some(bb)
1020                    if (direction.is_ascending() && bb < a_bucket)
1021                        || (!direction.is_ascending() && bb > a_bucket) =>
1022                {
1023                    // b is behind; skip it.
1024                    let _ = take_bucket_item(b.as_mut()).await?;
1025                }
1026                Some(_) => {
1027                    // Same bucket: subtract.
1028                    let (bid, a_bm) = take_bucket_item(a.as_mut()).await?;
1029                    let (_, b_bm) = take_bucket_item(b.as_mut()).await?;
1030                    let diff = a_bm - b_bm;
1031                    if !diff.is_empty() {
1032                        yield Watermarked::Item((bid, diff));
1033                    }
1034                }
1035            }
1036        }
1037    }
1038}

1040/// Convenience adapter: wrap a single raw `BucketStream` into a
1041/// `WatermarkedBucketStream` with one `Watermark(post_bucket)` after each
1042/// bucket plus one final at the range terminus on EOF.
1043///
1044/// The DNF eval pipeline uses `budget_limited_bucket_stream` instead;
1045/// this helper is for backend-side consumers (e.g. RocksDB
1046/// single-dimension scans) that want bucket-level output without the
1047/// full eval machinery.
1048pub fn buckets_with_watermarks<S>(
1049    stream: S,
1050    range: Range<u64>,
1051    bucket_size: u64,
1052    direction: ScanDirection,
1053) -> impl Stream<Item = Result<Watermarked<(u64, RoaringBitmap)>>> + Send + 'static
1054where
1055    S: Stream<Item = BucketItem> + Send + 'static,
1056{
1057    async_stream::try_stream! {
1058        if range.is_empty() {
1059            return;
1060        }
1061        futures::pin_mut!(stream);
1062        let mut last_emitted: Option<u64> = None;
1063        while let Some(item) = stream.next().await {
1064            let (bucket_id, bitmap) = item?;
1065            yield Watermarked::Item((bucket_id, bitmap));
1066            // Ascending = just past this bucket. Descending = this
1067            // bucket's low edge. Clamp to the request bounds — cursors
1068            // round-trip into subsequent requests with different ranges.
1069            let bucket_start = bucket_id.saturating_mul(bucket_size);
1070            let watermark = if direction.is_ascending() {
1071                bucket_start.saturating_add(bucket_size).min(range.end)
1072            } else {
1073                bucket_start.max(range.start)
1074            };
1075            last_emitted = Some(watermark);
1076            yield Watermarked::Watermark(watermark);
1077        }
1078        // Natural EOF: cap with a watermark at the range boundary so
1079        // handlers get an explicit "scan covered the range" signal.
1080        // Skip if a per-bucket watermark already exceeded it.
1081        let range_end = if direction.is_ascending() {
1082            range.end
1083        } else {
1084            range.start
1085        };
1086        let should_emit = match last_emitted {
1087            None => true,
1088            Some(prev) => {
1089                if direction.is_ascending() {
1090                    range_end > prev
1091                } else {
1092                    range_end < prev
1093                }
1094            }
1095        };
1096        if should_emit {
1097            yield Watermarked::Watermark(range_end);
1098        }
1099    }
1100}

1102/// Flatten marked bucket bitmaps into absolute member ids with
1103/// edge-bucket trimming against `range`. Watermarks pass through
1104/// unchanged.
1105pub fn flatten_watermarked_buckets<S>(
1106    stream: S,
1107    range: Range<u64>,
1108    bucket_size: u64,
1109    direction: ScanDirection,
1110) -> impl Stream<Item = Result<Watermarked<u64>>> + Send + 'static
1111where
1112    S: Stream<Item = Result<Watermarked<(u64, RoaringBitmap)>>> + Send + 'static,
1113{
1114    async_stream::try_stream! {
1115        if range.is_empty() {
1116            return;
1117        }
1118        let start_bucket = range.start / bucket_size;
1119        let end_bucket = (range.end - 1) / bucket_size;
1120        futures::pin_mut!(stream);
1121        while let Some(item) = stream.next().await {
1122            match item? {
1123                Watermarked::Watermark(p) => yield Watermarked::Watermark(p),
1124                Watermarked::Item((bucket_id, bitmap)) => {
1125                    let bucket_start = bucket_id * bucket_size;
1126                    let is_first = bucket_id == start_bucket;
1127                    let is_last = bucket_id == end_bucket;
1128                    let lo = if is_first {
1129                        (range.start - bucket_start) as u32
1130                    } else {
1131                        0
1132                    };
1133                    let hi = if is_last {
1134                        ((range.end - bucket_start).min(bucket_size)) as u32
1135                    } else {
1136                        bucket_size as u32
1137                    };
1138
1139                    if direction.is_ascending() {
1140                        for bit in bitmap.iter() {
1141                            if bit >= lo && bit < hi {
1142                                yield Watermarked::Item(bucket_start + bit as u64);
1143                            }
1144                        }
1145                    } else {
1146                        for bit in bitmap.iter().rev() {
1147                            if bit >= lo && bit < hi {
1148                                yield Watermarked::Item(bucket_start + bit as u64);
1149                            }
1150                        }
1151                    }
1152                }
1153            }
1154        }
1155    }
1156}

1158/// Outcome of `drain_pending_watermarks`: latest watermark consumed (if
1159/// any) AND any terminal error at the same head. Combinators apply the
1160/// watermark before propagating the error so mid-drain progress still
1161/// surfaces.
1162struct DrainOutcome {
1163    last_watermark: Option<u64>,
1164    error: Option<anyhow::Error>,
1165}
1166
1167/// Drain consecutive `Watermark` frames from the head of a
1168/// `Peekable<WatermarkedBucketStream>`. Combinators call this at the top of
1169/// each loop iteration so subsequent peeks see Items only.
1170async fn drain_pending_watermarks<S>(mut s: Pin<&mut Peekable<S>>) -> DrainOutcome
1171where
1172    S: Stream<Item = WatermarkedBucket>,
1173{
1174    let mut last: Option<u64> = None;
1175    loop {
1176        let action = match s.as_mut().peek().await {
1177            None => {
1178                return DrainOutcome {
1179                    last_watermark: last,
1180                    error: None,
1181                };
1182            }
1183            Some(Ok(Watermarked::Item(_))) => {
1184                return DrainOutcome {
1185                    last_watermark: last,
1186                    error: None,
1187                };
1188            }
1189            Some(Ok(Watermarked::Watermark(_))) => PeekAction::ConsumeWatermark,
1190            Some(Err(_)) => PeekAction::ConsumeError,
1191        };
1192        match action {
1193            PeekAction::ConsumeWatermark => match s.as_mut().next().await {
1194                Some(Ok(Watermarked::Watermark(p))) => last = Some(p),
1195                _ => unreachable!("peek confirmed Watermark"),
1196            },
1197            PeekAction::ConsumeError => match s.as_mut().next().await {
1198                Some(Err(e)) => {
1199                    return DrainOutcome {
1200                        last_watermark: last,
1201                        error: Some(e),
1202                    };
1203                }
1204                _ => unreachable!("peek confirmed Err"),
1205            },
1206        }
1207    }
1208}
1209
1210enum PeekAction {
1211    ConsumeWatermark,
1212    ConsumeError,
1213}

1215/// Consume the next bucket Item. Caller must have confirmed via
1216/// `peek_bucket` that the head is an Item — watermark or EOF here is a
1217/// logic error.
1218async fn take_bucket_item<S>(mut s: Pin<&mut Peekable<S>>) -> Result<(u64, RoaringBitmap)>
1219where
1220    S: Stream<Item = WatermarkedBucket>,
1221{
1222    match s.as_mut().next().await {
1223        Some(Ok(Watermarked::Item(it))) => Ok(it),
1224        Some(Ok(Watermarked::Watermark(_))) => {
1225            unreachable!("take_bucket_item called on Watermark — drain should have run first")
1226        }
1227        Some(Err(e)) => Err(e),
1228        None => unreachable!("take_bucket_item called on EOF — peek should have caught it"),
1229    }
1230}

1232/// Peek the next bucket_id. Caller MUST have run `drain_pending_watermarks`
1233/// in this loop iteration — a Watermark at the head here means the
1234/// drain step was skipped, which is a refactor bug rather than a
1235/// runtime condition. Returns `None` on EOF; surfaces errors via `?`.
1236async fn peek_bucket<S>(mut s: Pin<&mut Peekable<S>>) -> Result<Option<u64>>
1237where
1238    S: Stream<Item = WatermarkedBucket>,
1239{
1240    match s.as_mut().peek().await {
1241        None => Ok(None),
1242        Some(Ok(Watermarked::Item((b, _)))) => Ok(Some(*b)),
1243        Some(Ok(Watermarked::Watermark(_))) => {
1244            // Surface rather than silently consume: a stray WM here means
1245            // the per-iteration `drain_pending_watermarks` contract was
1246            // violated by a future refactor. Eating the WM would turn
1247            // that bug into "watermarks just stop appearing," which is
1248            // exactly the silent-progress-loss class this whole pipeline
1249            // is designed to avoid.
1250            bail!(
1251                "peek_bucket observed a stray Watermark — drain_pending_watermarks \
1252                 must run first in each combinator loop iteration"
1253            );
1254        }
1255        Some(Err(_)) => match s.as_mut().next().await {
1256            Some(Err(e)) => Err(e),
1257            _ => unreachable!("peek confirmed Err"),
1258        },
1259    }
1260}

1263where
1264    S: Stream<Item = WatermarkedBucket> + Unpin,
1265{
1266    futures::future::try_join_all(
1267        streams
1268            .iter_mut()
1269            .map(|stream| peek_bucket(Pin::new(stream))),

1274fn complete_peeks(peeks: Vec<Option<u64>>) -> Option<(Vec<u64>, u64)> {
1275    let mut buckets = Vec::with_capacity(peeks.len());
1276    let mut max_bucket = None;
1277
1278    for peek in peeks {
1279        let bucket = peek?;
1280        max_bucket = Some(max_bucket.map_or(bucket, |max: u64| max.max(bucket)));
1281        buckets.push(bucket);
1282    }
1283
1284    Some((buckets, max_bucket?))
1285}
